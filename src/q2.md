# Q2: Clean Electricity Share by Country, 2024

How do countries compare in their clean electricity share in 2024, and which have the highest and lowest shares?
Clean means renewables plus nuclear, as a percentage of domestic electricity generation.
Hover a country for its share and generation. Scroll to zoom and drag to pan the map; **Reset View** restores the original position.

```js
import * as topojson from "npm:topojson-client";

// Load Country Shapes (Atlas IDs are Zero-Padded Strings such as "004")
const world = await FileAttachment("data/countries-50m.json").json();
const features = topojson.feature(world, world.objects.countries).features
  .filter((d) => d.id !== "010"); // Antarctica Has No Study Data and Wastes Space
```

```js
// Load the Country-Year Table Untyped so iso_numeric Keeps its Leading Zeros, then Keep 2024 Countries
const countryYear = await FileAttachment("data/country_year.csv").csv();

const snapshot = countryYear
  .filter((d) => d.year === "2024" && d.country !== "World")
  .map((d) => ({
    ...d,
    cleanShare: d.clean_share_pct === "" ? null : +d.clean_share_pct,
    cleanTwh: d.clean_twh === "" ? null : +d.clean_twh,
    generationTwh: d.generation_twh === "" ? null : +d.generation_twh
  }));

const byId = new Map(snapshot.filter((d) => d.iso_numeric !== "").map((d) => [d.iso_numeric, d]));
```

```js
// Draw the Choropleth Map with a Natural Earth Projection and a Sequential Color Scale
const mapDims = {
  width: 900,
  height: 450,
  marginTop: 50,
  marginRight: 20,
  marginBottom: 70,
  marginLeft: 20
};

// No-Data Countries Get a Hatch Pattern so they Cannot be Mistaken for any Color on the Ramp
const noDataFill = "url(#q2-no-data-hatch)";
const oceanColor = "#e6eef5";
const outlineColor = "#8c8c8c";

// Start the Ramp at 15% of YlGnBu so a 0% Share Stays Visible Against the White Page and the Ocean
const color = d3.scaleSequential((t) => d3.interpolateYlGnBu(0.15 + 0.85 * t)).domain([0, 100]);

const projection = d3.geoNaturalEarth1()
  .fitSize([mapDims.width, mapDims.height], {type: "FeatureCollection", features});
const path = d3.geoPath(projection);

const mapSvg = d3.create("svg")
  .attr("width", mapDims.width + mapDims.marginLeft + mapDims.marginRight)
  .attr("height", mapDims.height + mapDims.marginTop + mapDims.marginBottom)
  .attr("viewBox", [0, 0, mapDims.width + mapDims.marginLeft + mapDims.marginRight, mapDims.height + mapDims.marginTop + mapDims.marginBottom])
  .attr("style", "max-width: 100%; height: auto; font: 11px sans-serif;");

mapSvg.append("text")
  .attr("x", (mapDims.width + mapDims.marginLeft + mapDims.marginRight) / 2)
  .attr("y", 25)
  .attr("text-anchor", "middle")
  .attr("font-size", 14)
  .attr("font-weight", 500)
  .text("Clean Share of Electricity Generation by Country, 2024");

// Hatch Pattern (Diagonal Lines on Off-White) for Countries without a Share
const mapDefs = mapSvg.append("defs");

const hatch = mapDefs.append("pattern")
  .attr("id", "q2-no-data-hatch")
  .attr("patternUnits", "userSpaceOnUse")
  .attr("width", 6)
  .attr("height", 6)
  .attr("patternTransform", "rotate(45)");

hatch.append("rect")
  .attr("width", 6)
  .attr("height", 6)
  .attr("fill", "#f4f4f4");

hatch.append("line")
  .attr("y2", 6)
  .attr("stroke", "#999999")
  .attr("stroke-width", 1.5);

// Clip the Zoomed Map to its Frame
mapDefs.append("clipPath")
  .attr("id", "q2-map-clip")
  .append("rect")
  .attr("x", mapDims.marginLeft)
  .attr("y", mapDims.marginTop)
  .attr("width", mapDims.width)
  .attr("height", mapDims.height);

// Ocean Background (Static, Behind the Zooming Map)
mapSvg.append("rect")
  .attr("x", mapDims.marginLeft)
  .attr("y", mapDims.marginTop)
  .attr("width", mapDims.width)
  .attr("height", mapDims.height)
  .attr("fill", oceanColor)
  .attr("stroke", outlineColor)
  .attr("stroke-width", 0.5);

const mapChart = mapSvg.append("g")
  .attr("clip-path", "url(#q2-map-clip)")
  .append("g")
  .attr("transform", `translate(${mapDims.marginLeft}, ${mapDims.marginTop})`);

// Country Shapes (Grey = No Matched Data)
mapChart.selectAll("path.country")
  .data(features)
  .join("path")
  .attr("class", "country")
  .attr("d", path)
  .attr("fill", (d) => {
    const row = byId.get(d.id);
    return row?.cleanShare == null ? noDataFill : color(row.cleanShare);
  })
  .attr("stroke", outlineColor)
  .attr("stroke-width", 0.4)
  .style("vector-effect", "non-scaling-stroke")
  .append("title")
  .text((d) => {
    const row = byId.get(d.id);
    if (!row || row.cleanShare == null) return `${d.properties.name}\nNo Matched Data`;
    return `${row.country}\nClean Share: ${d3.format(".1f")(row.cleanShare)}%\nClean: ${d3.format(",.1f")(row.cleanTwh)} TWh\nGeneration: ${d3.format(",.1f")(row.generationTwh)} TWh`;
  });

// Zoom and Pan (Clipped by the Viewport via the Outer SVG)
const zoom = d3.zoom()
  .scaleExtent([1, 8])
  .on("zoom", (event) => {
    mapChart.attr("transform", `translate(${mapDims.marginLeft}, ${mapDims.marginTop}) ${event.transform}`);
  });

mapSvg.call(zoom);

// Reset View Button that Animates the Map Back to its Original Zoom and Position
const resetButton = document.createElement("button");
resetButton.textContent = "⟲ Reset View";
resetButton.onclick = () => mapSvg.transition().duration(500).call(zoom.transform, d3.zoomIdentity);

// Legend: Gradient Sampled from the Color Scale
const legendW = 300;
const legendH = 12;
const legendX = (mapDims.width + mapDims.marginLeft + mapDims.marginRight) / 2 - legendW / 2;
const legendY = mapDims.marginTop + mapDims.height + 25;

const gradient = mapSvg.append("defs")
  .append("linearGradient")
  .attr("id", "q2-share-gradient");

d3.range(0, 1.01, 0.1).forEach((t) => {
  gradient.append("stop")
    .attr("offset", `${t * 100}%`)
    .attr("stop-color", color(t * 100));
});

const legend = mapSvg.append("g")
  .attr("transform", `translate(${legendX}, ${legendY})`);

legend.append("rect")
  .attr("width", legendW)
  .attr("height", legendH)
  .attr("fill", "url(#q2-share-gradient)")
  .attr("stroke", outlineColor)
  .attr("stroke-width", 0.5);

legend.append("g")
  .attr("transform", `translate(0, ${legendH})`)
  .call(d3.axisBottom(d3.scaleLinear().domain([0, 100]).range([0, legendW])).ticks(5))
  .call((g) => g.select(".domain").remove());

legend.append("text")
  .attr("x", legendW / 2)
  .attr("y", legendH + 32)
  .attr("text-anchor", "middle")
  .text("Clean Share of Electricity Generation (%)");

// No Data Swatch
const noData = mapSvg.append("g")
  .attr("transform", `translate(${mapDims.marginLeft}, ${legendY - 4})`);

noData.append("rect")
  .attr("width", 16)
  .attr("height", 16)
  .attr("fill", noDataFill)
  .attr("stroke", outlineColor)
  .attr("stroke-width", 0.5);

noData.append("text")
  .attr("x", 24)
  .attr("y", 12)
  .text("No Matched Data");

display(resetButton);
display(mapSvg.node());
```

## Highest and Lowest Shares

```js
// Create the Slider for How Many Countries Appear in Each Panel
const countInput = Inputs.range([3, 20], {label: "Countries per Panel", step: 1, value: 10});
const count = view(countInput);
```

```js
// Rank Countries with a Known Share (Ties Broken by Country Name)
const known = snapshot.filter((d) => d.cleanShare != null);

const highest = known
  .slice()
  .sort((a, b) => d3.descending(a.cleanShare, b.cleanShare) || d3.ascending(a.country, b.country))
  .slice(0, count);

const lowest = known
  .slice()
  .sort((a, b) => d3.ascending(a.cleanShare, b.cleanShare) || d3.ascending(a.country, b.country))
  .slice(0, count);
```

```js
// Draw the Two Ranking Panels on a Shared 0-100% Scale
const dims = {
  panelWidth: 380,
  rowHeight: 24,
  gap: 30,
  marginTop: 60,
  marginRight: 20,
  marginBottom: 50,
  marginLeft: 160
};

const innerHeight = count * dims.rowHeight;
const totalWidth = dims.marginLeft + dims.panelWidth + dims.gap + dims.marginLeft + dims.panelWidth + dims.marginRight;
const totalHeight = innerHeight + dims.marginTop + dims.marginBottom;

const svg = d3.create("svg")
  .attr("width", totalWidth)
  .attr("height", totalHeight)
  .attr("viewBox", [0, 0, totalWidth, totalHeight])
  .attr("style", "max-width: 100%; height: auto; font: 11px sans-serif;");

svg.append("text")
  .attr("x", totalWidth / 2)
  .attr("y", 22)
  .attr("text-anchor", "middle")
  .attr("font-size", 14)
  .attr("font-weight", 500)
  .text("Highest and Lowest Clean Shares, 2024");

const x = d3.scaleLinear().domain([0, 100]).range([0, dims.panelWidth]);
const pct = d3.format(".1f");

// Draw One Panel (Bars, Labels, Axis and Heading)
function drawPanel(rows, {title, fill, left}) {
  const y = d3.scaleBand()
    .domain(rows.map((d) => d.country))
    .range([0, innerHeight])
    .padding(0.25);

  const g = svg.append("g")
    .attr("transform", `translate(${left}, ${dims.marginTop})`);

  g.append("text")
    .attr("x", dims.panelWidth / 2)
    .attr("y", -18)
    .attr("text-anchor", "middle")
    .attr("font-size", 12)
    .text(title);

  // Vertical Gridlines
  g.append("g")
    .selectAll("line")
    .data(x.ticks(4))
    .join("line")
    .attr("x1", (d) => x(d))
    .attr("x2", (d) => x(d))
    .attr("y1", 0)
    .attr("y2", innerHeight)
    .attr("stroke", "currentColor")
    .attr("stroke-opacity", 0.1);

  // Bars
  g.append("g")
    .selectAll("rect")
    .data(rows, (d) => d.country)
    .join("rect")
    .attr("x", 0)
    .attr("y", (d) => y(d.country))
    .attr("width", (d) => x(d.cleanShare))
    .attr("height", y.bandwidth())
    .attr("fill", fill)
    .append("title")
    .text((d) => `${d.country}\nClean Share: ${pct(d.cleanShare)}%\nGeneration: ${d3.format(",.1f")(d.generationTwh)} TWh`);

  // Value Labels Just Beyond the Bar End
  g.append("g")
    .selectAll("text")
    .data(rows, (d) => d.country)
    .join("text")
    .attr("x", (d) => x(d.cleanShare) + 4)
    .attr("y", (d) => y(d.country) + y.bandwidth() / 2)
    .attr("dy", "0.35em")
    .attr("font-size", 10)
    .text((d) => `${pct(d.cleanShare)}%`);

  // Country Labels at the Left Edge of the Panel
  g.append("g")
    .selectAll("text")
    .data(rows, (d) => d.country)
    .join("text")
    .attr("x", -8)
    .attr("y", (d) => y(d.country) + y.bandwidth() / 2)
    .attr("dy", "0.35em")
    .attr("text-anchor", "end")
    .text((d) => d.country);

  // X Axis
  g.append("g")
    .attr("transform", `translate(0, ${innerHeight})`)
    .call(d3.axisBottom(x).ticks(4))
    .call((a) => a.select(".domain").remove());

  g.append("text")
    .attr("x", dims.panelWidth / 2)
    .attr("y", innerHeight + 38)
    .attr("text-anchor", "middle")
    .text("Clean Share (%)");
}

drawPanel(highest, {title: "Highest Shares", fill: "#24857a", left: dims.marginLeft});
drawPanel(lowest, {title: "Lowest Shares", fill: "#b5614a", left: dims.marginLeft + dims.panelWidth + dims.gap + dims.marginLeft});

display(svg.node());
```
