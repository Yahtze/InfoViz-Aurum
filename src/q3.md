# Q3: Largest Clean-Share Changes, 2000–2024

Which countries gained or lost the most clean electricity share between 2000 and 2024?
Bars show the change in clean share (renewables + nuclear) in **percentage points**.
Only the 108 countries with endpoint data and at least 10 TWh of generation in 2024 are compared.
Hover a bar for the 2000 and 2024 shares and the change in fossil generation.

```js
// Create the Full Screen Panel Once, so Changing a Control Redraws the Chart Without Leaving Full Screen
import {fullscreenPanel} from "./components/fullscreen.js";

const panel = fullscreenPanel();
display(panel.element);
```

```js
// Load the Comparison Table (One Row per Compared Country)
const comparison = await FileAttachment("data/comparison_2000_2024.csv").csv({typed: true});
```

```js
// Create the Slider for How Many Countries Appear in Each Tail
const countInput = Inputs.range([3, 20], {label: "Countries per Panel", step: 1, value: 10});
panel.addControl(countInput);
const count = Generators.input(countInput);
```

```js
// Pick the Largest Increases and Decreases (Only Positive Gains and Negative Losses Appear)
const increases = comparison
  .filter((d) => d.d_clean_share_pp > 0)
  .sort((a, b) => d3.descending(a.d_clean_share_pp, b.d_clean_share_pp) || d3.ascending(a.country, b.country))
  .slice(0, count);

const decreases = comparison
  .filter((d) => d.d_clean_share_pp < 0)
  .sort((a, b) => d3.ascending(a.d_clean_share_pp, b.d_clean_share_pp) || d3.ascending(a.country, b.country))
  .slice(0, count);
```

```js
// Draw the Two Diverging Bar Panels (Shared X Scale so Bar Lengths are Comparable)
const dims = {
  panelWidth: 380,
  rowHeight: 24,
  gap: 30,
  marginTop: 60,
  marginRight: 20,
  marginBottom: 50,
  marginLeft: 120
};

const innerHeight = count * dims.rowHeight;
const totalWidth = dims.marginLeft + dims.panelWidth + dims.gap + dims.marginLeft + dims.panelWidth + dims.marginRight;

const svg = d3.create("svg")
  .attr("width", totalWidth)
  .attr("height", innerHeight + dims.marginTop + dims.marginBottom)
  .attr("viewBox", [0, 0, totalWidth, innerHeight + dims.marginTop + dims.marginBottom])
  .attr("style", "max-width: 100%; height: auto; font: 11px sans-serif;");

svg.append("text")
  .attr("x", totalWidth / 2)
  .attr("y", 22)
  .attr("text-anchor", "middle")
  .attr("font-size", 14)
  .attr("font-weight", 500)
  .text("Largest Changes in Clean Electricity Share, 2000–2024");

// Symmetric Domain Shared by Both Panels
const maxChange = d3.max([...increases, ...decreases], (d) => Math.abs(d.d_clean_share_pp));
const x = d3.scaleLinear()
  .domain([-maxChange, maxChange])
  .nice()
  .range([0, dims.panelWidth]);

const signed = d3.format("+.1f");
const pct = d3.format(".1f");

// Draw One Panel (Bars, Labels, Axis and Heading)
function drawPanel(rows, {title, color, left}) {
  const y = d3.scaleBand()
    .domain(rows.map((d) => d.country))
    .range([0, innerHeight])
    .padding(0.25);

  const g = svg.append("g")
    .attr("transform", `translate(${left}, ${dims.marginTop})`);

  g.append("text")
    .attr("x", dims.panelWidth / 2)
    .attr("y", -18)
    .attr("text-anchor", "middle")
    .attr("font-size", 12)
    .text(title);

  // Vertical Gridlines
  g.append("g")
    .selectAll("line")
    .data(x.ticks(6))
    .join("line")
    .attr("x1", (d) => x(d))
    .attr("x2", (d) => x(d))
    .attr("y1", 0)
    .attr("y2", innerHeight)
    .attr("stroke", "currentColor")
    .attr("stroke-opacity", 0.1);

  // Bars Grow from Zero in the Direction of the Change
  g.append("g")
    .selectAll("rect")
    .data(rows, (d) => d.country)
    .join("rect")
    .attr("x", (d) => x(Math.min(0, d.d_clean_share_pp)))
    .attr("y", (d) => y(d.country))
    .attr("width", (d) => Math.abs(x(d.d_clean_share_pp) - x(0)))
    .attr("height", y.bandwidth())
    .attr("fill", color)
    .append("title")
    .text((d) => `${d.country}\n2000: ${pct(d.clean_share_pct_2000)}%\n2024: ${pct(d.clean_share_pct_2024)}%\nChange: ${signed(d.d_clean_share_pp)} pp\nFossil Generation: ${signed(d.d_fossil_twh)} TWh`);

  // Value Labels Just Beyond the Bar End
  g.append("g")
    .selectAll("text")
    .data(rows, (d) => d.country)
    .join("text")
    .attr("x", (d) => x(d.d_clean_share_pp) + (d.d_clean_share_pp >= 0 ? 4 : -4))
    .attr("y", (d) => y(d.country) + y.bandwidth() / 2)
    .attr("dy", "0.35em")
    .attr("text-anchor", (d) => (d.d_clean_share_pp >= 0 ? "start" : "end"))
    .attr("font-size", 10)
    .text((d) => signed(d.d_clean_share_pp));

  // Zero Baseline
  g.append("line")
    .attr("x1", x(0))
    .attr("x2", x(0))
    .attr("y1", 0)
    .attr("y2", innerHeight)
    .attr("stroke", "currentColor");

  // Country Labels at the Left Edge of the Panel
  g.append("g")
    .selectAll("text")
    .data(rows, (d) => d.country)
    .join("text")
    .attr("x", -8)
    .attr("y", (d) => y(d.country) + y.bandwidth() / 2)
    .attr("dy", "0.35em")
    .attr("text-anchor", "end")
    .text((d) => d.country);

  // X Axis
  g.append("g")
    .attr("transform", `translate(0, ${innerHeight})`)
    .call(d3.axisBottom(x).ticks(6))
    .call((a) => a.select(".domain").remove());

  g.append("text")
    .attr("x", dims.panelWidth / 2)
    .attr("y", innerHeight + 38)
    .attr("text-anchor", "middle")
    .text("Change in Clean Share (Percentage Points)");
}

drawPanel(increases, {title: "Largest Increases", color: "#24857a", left: dims.marginLeft});
drawPanel(decreases, {title: "Largest Decreases", color: "#b5614a", left: dims.marginLeft + dims.panelWidth + dims.gap + dims.marginLeft});

panel.setChart(svg.node());
```
